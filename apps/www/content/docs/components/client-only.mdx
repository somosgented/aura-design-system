```bash
pnpm dlx shadcn@latest add @aura/client-only
```



### Manual

<Steps>
  <Step>
    Copy and paste the ClientOnly component into your `components/ui/ClientOnly.tsx` file.

```tsx
"use client";

import * as React from "react";

function ClientOnly({
  children,
  fallback = null,
}: {
  children: React.ReactNode;
  fallback?: React.ReactNode;
}) {
  const [ready, setReady] = React.useState(false);
  React.useEffect(() => setReady(true), []);
  return ready ? children : fallback;
}

export { ClientOnly };

```
  </Step>
</Steps>
